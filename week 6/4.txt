class UserProfile:
    def __init__(self, username="", email=""):
        self.username = username
        self.email = email

    def get_username(self):
        return self.username

    def get_email(self):
        return self.email

    def set_username(self, username):
        self.username = username

    def set_email(self, email):
        self.email = email

user = UserProfile()
user.set_username(input("Enter initial username: "))
user.set_email(input("Enter initial email: "))

while True:
    print("\n--- Menu ---")
    print("1. View Data")
    print("2. Modify Username")
    print("3. Modify Email")
    print("4. Exit")
    
    choice = input("Choose an option (1-4): ")

    if choice == "1":
        print(f"\nUsername: {user.get_username()}")
        print(f"Email   : {user.get_email()}")
    elif choice == "2":
        new_name = input("Enter new username: ")
        user.set_username(new_name)
        print("Username updated!")
    elif choice == "3":
        new_email = input("Enter new email: ")
        user.set_email(new_email)
        print("Email updated!")
    elif choice == "4":
        print("Exiting program.")
        break
    else:
        print("Invalid choice. Please select 1-4.")
