def grade_point(grade):
    if grade == "A":
        return 4.00
    elif grade == "A-":
        return 3.75
    elif grade == "B+":
        return 3.50
    elif grade == "B":
        return 3.00
    elif grade == "B-":
        return 2.75
    elif grade == "C+":
        return 2.50
    elif grade == "C":
        return 2.00
    elif grade == "C-":
        return 1.75
    elif grade == "D":
        return 1.50
    else:
        return 1.20

total = 0
ammount = 0

while True:
    grade = input("Enter Grade Category (Press Enter to Stop): ").upper().strip()
    
    if grade == "":
        break
        
    total = total + grade_point(grade)
    ammount = ammount + 1

if ammount > 0:
    rate = total / ammount


    if rate >= 3.88:
        letter = "A"
    elif rate >= 3.63:
        letter = "A-"
    elif rate >= 3.25:
        letter = "B+"
    elif rate >= 2.88:
        letter = "B"
    elif rate >= 2.63:
        letter = "B-"
    elif rate >= 2.25:
        letter = "C+"
    elif rate >= 1.88:
        letter = "C"
    elif rate >= 1.63:
        letter = "C-"
    elif rate >= 1.35:
        letter = "D"
    else:
        letter = "E"

    print(f"The average grade is {rate:.2f} with a designation of {letter}")
else:
    print("No grades were entered.")
